# The hidden preacher: peasants hide a Hussite preacher from the bishop's riders.
name     The hidden preacher
card     $NwHus00
picture  XHIDDEN2
where    wild
terrain  farmland woods hills plains
when     any
weight   1

[start]
block    0
chosen   bestskillsum(Religion, Latin)
option 0                                   # dispute Scripture with him
  who    bestskillsum(Religion, Latin)
  chance clamp(5, 90, skill(who, Religion) + skill(who, Latin) / 2 + attr(who, Int) / 4 - 20)
  ok     chosen who; text 1; practice who, Religion, 1, 10, 0; practice who, Latin, 1, 10, 0; attr who, DF, 2; hours 2; exit
  fail   chosen who; text 2; practice who, Religion, 0, 10, 0; attr who, DF, -2; hours 2; exit
option 1                                   # report him to the bishop's men in town
  chance assured
  ok     money1 72; text 3; gain money1; reputation 1, 3; skill party, Virtue, -1; hours 3; tocity
option 2                                   # lead him to safety by the back ways
  who    party
  chance clamp(5, 90, avgskill(Woodwise) + avgskill(Stealth) / 2 + avgagility / 3)
  ok     text 4; hours 6; practice party, Woodwise, 1, 10, 0; practice party, Virtue, 1, 10, 0; attr party, DF, -1; exit
  fail   money1 min(money, 240); text 5; pay money1; reputation -1, -3; hours 4; exit
option 3                                   # keep silent and go on
  chance none
  ok     text 6; exit
