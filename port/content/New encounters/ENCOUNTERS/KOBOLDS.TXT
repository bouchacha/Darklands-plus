# Kobolds at the mine: something comes up out of the deep galleries at night.
name     Kobolds at the mine
card     $NwKob00
picture  XMINER4
terrain  hills mountains pinewood
when     day
weight   2

[start]
do       money1 96 + lore * 12
block    0
saints   0x0d, 0x60, 0x80, -1            # Barbara, Michael, Vitus
option 0                                   # keep watch at the mine mouth
  chance combat
  ok     goto deep
option 1 saint                             # bless the mine
  ok     money1 money1 / 2; gain money1; text 2; attr party, DF, 2; exit
  fail   text 3
option 2                                   # leave the miners to their troubles
  chance none
  ok     text 4; exit

[deep]
picture  XKBOLD2
do       text 1; until 22; battle Clearing, Kobold, lore / 4 + 1, random(4) + 3
         if result == Won then { gain money1; text 5; hours 3; fame 1; exit }
         if result == Fled then { rejoin; text 6; until 5; exit }
         loseitems party; text 7; until 5; exit
