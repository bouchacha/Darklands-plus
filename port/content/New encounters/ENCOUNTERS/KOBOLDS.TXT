# Kobolds in the mine: the miners' deep galleries are haunted.
name     Kobolds in the mine
card     $NwKob00
picture  XMINER4
terrain  hills mountains pinewood
when     day
weight   2

[start]
do       money1 120 + lore * 24
block    0
saints   0x0d, 0x60, 0x68, 0x1a          # Barbara, Michael, Patrick, Columba
option 0                                   # go down into the mine
  chance combat
  ok     goto deep
option 1 saint                             # bless the mine
  ok     money1 money1 / 2; gain money1; text 2; attr party, DF, 2; exit
  fail   text 3
option 2                                   # leave the miners to their troubles
  chance none
  ok     text 4; exit

[deep]
picture  XKBOLD2
do       text 1; battle Clearing, Kobold, lore / 4 + 1, random(4) + 3
         if result == Won then { gain money1; text 5; hours 3; fame 1; exit }
         if result == Fled then { rejoin; text 6; hours 2; exit }
         loseitems party; text 7; hours 4; exit
