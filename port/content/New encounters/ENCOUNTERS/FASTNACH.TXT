# The Fools' Court: carnival masks haul you before a mock court for the crime of walking sober.
name     The Fools' Court
card     $NdFas00
picture  MS120
where    street
when     day
weight   1
only     month <= 2

[start]
do       money1 36; money2 24
block    0
chosen   bestattr(Cha)
option 0                                   # plead guilty and pay the fine in beer
  need   money >= money1
  chance none
  ok     text 1; pay money1; reputation 1, 2; attr party, End, 2; hours 1; exit
option 1                                   # plead your case in the court's own manner
  who    bestattr(Cha)
  chance clamp(5, 90, (2 * attr(who, Cha) + skill(who, SpeakCommon)) / 3 - 5)
  ok     chosen who; text 2; practice who, SpeakCommon, 1, 10, 0; goto king
  fail   chosen who; text 3; attr party, End, -3; reputation 0, 1; practice who, SpeakCommon, 0, 10, 0; exit
option 2                                   # break away from the masks
  who    party
  chance clamp(10, 90, avgagility + 15)
  ok     text 4; exit
  fail   text 5; attr party, End, -3; reputation -1, -1; exit

[king]
block    6
option 0                                   # scatter largesse to your subjects
  need   money >= money2
  chance none
  ok     text 7; pay money2; fame 1; reputation 1, 2; exit
option 1                                   # decree that the judge be ducked
  who    party
  chance clamp(10, 90, maxattr(Str) + avgattr(Str) / 2 - 10)
  ok     text 8; reputation 1, 3; attr party, End, -2; exit
  fail   text 9; attr party, End, -3; reputation 0, 1; exit
option 2                                   # abdicate in favour of the butcher's goat
  chance none
  ok     text 10; reputation 1, 1; exit
