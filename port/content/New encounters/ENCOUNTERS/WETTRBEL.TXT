# The weather bell: hail over the barley, a chapel bell to ring against it, and lightning for the ringer.
name     The weather bell
card     $NhWbl00
picture  MONKBELL
where    wild
terrain  hills mountains
when     day
weight   1
only     month >= 6 and month <= 8

[start]
block    0
chosen   bestattrsum(Str, End)
saints   0x0d, 0x33, 0x01, 0x60          # Barbara, Florian, Agatha, Michael
option 0                                   # run up and ring the bell (strength and endurance)
  who    bestattrsum(Str, End)
  chance clamp(10, 90, (attr(who, Str) + attr(who, End)) * 2 / 3 + 10)
  ok     chosen who; text 1; hours 2; attr who, End, -3; reputation 1, 3; exit
  fail   chosen who; text 2; hours 2; injure who, 1, 1, 10; exit
option 1                                   # rig the rope to ring from the shelter of the stair (Artifice)
  who    bestskill(Artifice)
  chance clamp(5, 90, skill(who, Artifice) + attr(who, Int) / 3 - 5)
  ok     chosen who; text 3; hours 2; practice who, Artifice, 1, 10, 0; reputation 1, 3; exit
  fail   chosen who; text 4; hours 2; practice who, Artifice, 0, 10, 0; exit
option 2 saint
  ok     text 5; hours 1; reputation 1, 2; exit
  fail   text 6
option 3                                   # shelter in the chapel, beside the rope
  chance none
  ok     text 7; hours 2; reputation -1, -2; exit
option 4                                   # hurry on before it breaks
  chance none
  ok     text 8; hours 1; attr party, End, -2; exit
