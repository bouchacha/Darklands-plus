# The guild quarrel: two guilds' processions meet in a narrow street, and neither will give way.
name     The guild quarrel
card     $NdGld00
picture  XMOB
where    street
when     day
weight   1

[start]
block    0
chosen   bestattrskill(Cha, Streetwise)
option 0                                   # ask the oldest folk in the street what custom says
  who    bestattrskill(Cha, Streetwise)
  chance clamp(5, 90, attr(who, Cha) / 2 + skill(who, Streetwise) - 5)
  ok     chosen who; text 1; reputation 1, 3; practice who, Streetwise, 1, 10, 0; hours 1; exit
  fail   chosen who; text 2; practice who, Streetwise, 0, 10, 0; goto brawl
option 1                                   # bid them cast lots, as the Apostles did
  who    bestskill(Religion)
  chance clamp(5, 90, skill(who, Religion) + attr(who, Cha) / 3 - 5)
  ok     chosen who; text 3; reputation 1, 2; practice who, Religion, 1, 10, 0; exit
  fail   chosen who; text 4; practice who, Religion, 0, 10, 0; goto brawl
option 2                                   # go the long way round
  chance none
  ok     text 5; hours 1; exit

[brawl]
block    6
option 0                                   # drag the beadle out of the press
  who    bestattrsum(Str, Agl)
  chance clamp(10, 90, attr(who, Str) + attr(who, Agl) / 2 - 10)
  ok     chosen who; text 7; reputation 1, 2; exit
  fail   chosen who; text 8; injure who, 1, 1, 5; exit
option 1                                   # drive them apart with a banner pole
  who    bestskill(Polearms)
  chance clamp(5, 90, skill(who, Polearms) + attr(who, Str) / 3 - 10)
  ok     chosen who; text 9; reputation 1, 3; practice who, Polearms, 1, 10, 0; exit
  fail   chosen who; text 10; injure who, 1, 1, 10; reputation -1, -2; practice who, Polearms, 0, 10, 0; exit
option 2                                   # slip away while they are busy
  who    party
  chance clamp(10, 90, avgskill(Stealth) + avgagility * 2 / 3 - 5)
  ok     text 11; practice party, Stealth, 1, 10, 0; exit
  fail   text 12; reputation -1, -2; practice party, Stealth, 0, 10, 0; exit
