# The lepers' clapper: lepers begging alms by the road.
name     The lepers' clapper
card     $NwLep00
picture  MS016
where    wild
terrain  road farmland plains
when     day
weight   1

[start]
do       money1 size * 12
block    0
chosen   bestskill(Healing)
saints   0x55, 0x3a, 0x5d, 0x73          # Lazarus, Giles, Martin of Tours, Roch
option 0                                   # alms
  need   money >= money1
  chance none
  ok     text 1; pay money1; attr party, DF, 2; practice party, Virtue, 1, 5, -1; exit
option 1                                   # tend their sores (a fever if it goes badly)
  who    bestskill(Healing)
  chance clamp(5, 90, skill(who, Healing) + attr(who, Int) / 3 - 10)
  ok     chosen who; text 2; hours 2; attr party, DF, 2; attr who, DF, 3; practice who, Healing, 1, 10, 0; exit
  fail   chosen who; text 3; until 6; attr who, End, -5; attr who, DF, 2; practice who, Healing, 0, 10, 0; exit
option 2 saint
  ok     text 4; attr party, DF, 2; exit
  fail   text 5
option 3                                   # drive them off
  chance none
  ok     text 6; attr party, DF, -5; exit
option 4                                   # hurry past
  chance none
  ok     text 7; exit
