# The guildhall: shapes at the weavers' hall by night, and what they came to take back.
name     The guildhall
card     $NnGld00
picture  XCREEP
where    street
when     night
weight   1

[start]
block    0
chosen   bestattrskill(Per, Stealth)
option 0                                   # raise the alarm
  chance none
  ok     text 1; reputation 1, 2; hours 1; exit
option 1                                   # creep up and see who they are
  who    bestattrskill(Per, Stealth)
  chance clamp(5, 90, skill(who, Stealth) + attr(who, Per) / 3 - 5)
  ok     chosen who; practice who, Stealth, 1, 10, 0; goto weavers
  fail   chosen who; text 2; practice who, Stealth, 0, 10, 0; hours 1; exit
option 2                                   # put a shot into the shutter beside them (Bows, Missile)
  need   has(party, 0x1a) or has(party, 0x1b) or has(party, 0x1c) or has(party, 0x1d) or has(party, 0x1e)
  who    bestskill2(Bows, Missile)
  chance clamp(5, 90, max(skill(who, Bows), skill(who, Missile)) + attr(who, Per) / 3 - 10)
  ok     chosen who; text 3; reputation 1, 3
         if skill(who, Bows) >= skill(who, Missile) then practice who, Bows, 1, 10, 0 else practice who, Missile, 1, 10, 0
         exit
  fail   chosen who; text 4; reputation -1, -3; exit
option 3                                   # keep walking
  chance none
  ok     text 5; exit

[weavers]
block    6
option 0                                   # help them with the lock (Artifice)
  who    bestattrskill(Agl, Artifice)
  chance clamp(5, 90, skill(who, Artifice) + attr(who, Agl) / 3 - 10)
  ok     chosen who; text 7; attr party, DF, 1; practice who, Artifice, 1, 10, 0; hours 1; exit
  fail   chosen who; text 8; reputation -1, -3; practice who, Artifice, 0, 10, 0; hours 1; exit
option 1                                   # call the watch on them
  chance none
  ok     text 9; reputation 1, 3; attr party, DF, -2; hours 1; exit
option 2                                   # seize them yourselves
  chance combat
  ok     goto fight
option 3                                   # leave them to it
  chance none
  ok     text 10; exit

[fight]
do       battle SideStreet, Thug, lore / 4 + 1, random(2) + 3
         if result == Won then { text 11; reputation 1, 3; hours 1; exit }
         if result == Fled then { rejoin; text 12; reputation -1, -2; exit }
         text 13; until 5; reputation -1, -2; exit
