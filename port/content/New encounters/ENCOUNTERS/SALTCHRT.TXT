# The salt-master's charter: a salt convoy stopped by a town's riders, and a charter no one can read.
name     The salt-master's charter
card     $NrSlt00
picture  GUARD1
where    wild
terrain  road plains farmland
when     day
weight   1

[start]
block    0
chosen   bestskill(ReadWrite)
option 0                                   # read the charter out (ReadWrite)
  who    bestskill(ReadWrite)
  chance clamp(5, 90, skill(who, ReadWrite) + skill(who, Latin) / 4 + attr(who, Int) / 3 - 10)
  ok     chosen who; practice who, ReadWrite, 1, 10, 0; goto charter
  fail   chosen who; text 2; hours 1; practice who, ReadWrite, 0, 10, 0; disable 0
option 1                                   # swear to it unread (SpeakCommon)
  who    bestattr(Cha)
  chance clamp(5, 90, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5 - 10)
  ok     chosen who; money1 random(24) + 24; text 3; gain money1; attr who, DF, -2; practice who, SpeakCommon, 1, 10, 0; exit
  fail   chosen who; text 4; reputation -1, -3; practice who, SpeakCommon, 0, 10, 0; exit
option 2                                   # leave them to it
  chance none
  ok     text 5; exit

[charter]
block    1
option 0                                   # the truth, date and all
  chance none
  ok     text 6; reputation 1, 3; practice chosen, Virtue, 1, 10, 0; exit
option 1                                   # never come to the date (SpeakCommon)
  who    chosen
  chance clamp(5, 90, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5)
  ok     money1 random(36) + 36; text 7; gain money1; attr party, DF, -3; practice who, SpeakCommon, 1, 10, 0; exit
  fail   text 8; reputation -1, -3; attr who, DF, -1; practice who, SpeakCommon, 0, 10, 0; exit
option 2                                   # tell the salt-master first, and broker a bargain (Streetwise)
  who    bestskill(Streetwise)
  chance clamp(5, 90, skill(who, Streetwise) + attr(who, Cha) / 3)
  ok     chosen who; money1 random(24) + 12; text 9; gain money1; reputation 1, 2; practice who, Streetwise, 1, 10, 0; exit
  fail   chosen who; text 10; hours 1; practice who, Streetwise, 0, 10, 0; exit
