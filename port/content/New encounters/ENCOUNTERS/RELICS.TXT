# The relic seller: a holy bone for sale, cheap.
name     The relic seller
card     $NwRel00
picture  XUGLYMAN
terrain  road farmland plains
when     day
weight   1

[start]
do       money1 clamp(24, 240, money / 10)
block    0
chosen   bestskill(Religion)
option 0                                   # buy the relic
  need   money >= money1
  chance none
  ok     pay money1; text 1; exit
option 1                                   # examine it closely
  who    bestskill(Religion)
  chance clamp(5, 95, skill(who, Religion) + attr(who, Int) / 3)
  ok     chosen who; text 2; practice who, Religion, 1, 10, 0; attr party, DF, 1; exit
  fail   chosen who; text 3; practice who, Religion, 0, 10, 0; money1 money1 / 2; disable 1
option 2                                   # send him on his way
  chance none
  ok     text 4; exit
