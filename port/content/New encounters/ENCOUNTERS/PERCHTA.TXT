# The goose-foot track: in the dark of the year, Frau Perchta's tracks in the new snow, round every sleeper's head.
name     The goose-foot track
card     $NkPer00
picture  XMS056
where    camp
when     any
weight   1
only     (month == 12 or month == 1) and size >= 2

[start]
block    0
chosen   worstskill(Virtue)                    # who slept through the watch
option 0                                   # look in your boots to see how she judged you (the party's Virtue)
  who    party
  chance clamp(10, 90, avgskill(Virtue) + 30)
  ok     money1 random(12) + size * 6; text 1; gain money1; attr chosen, End, -2; exit
  fail   text 2; attr party, End, -2; exit
option 1                                   # sweep the camp as clean as a goodwife's kitchen (time)
  chance assured
  ok     text 3; hours 2; practice chosen, Virtue, 1, 10, 0; exit
option 2                                   # pray, and chalk the Three Kings' crosses on a tree (Religion)
  who    bestskill(Religion)
  chance clamp(10, 90, skill(who, Religion) + attr(who, Cha) / 3 + 5)
  ok     chosen2 who; text 4; attr party, DF, 2; practice who, Religion, 1, 10, 0; exit
  fail   chosen2 who; text 5; attr chosen, End, -4; practice who, Religion, 0, 10, 0; exit
option 3                                   # laugh at old wives' tales and march
  chance none
  ok     text 6; attr chosen, End, -3; exit
