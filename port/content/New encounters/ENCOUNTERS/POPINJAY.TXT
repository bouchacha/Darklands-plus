# The popinjay shoot: the crossbowmen's guild shoots at the wooden bird, and the new king must stand the beer.
name     The popinjay shoot
card     $NdPop00
picture  XBOWYER
where    street
when     day
weight   1
only     month >= 5 and month <= 9

[start]
do       money1 120; money2 random(36) + 48
block    0
chosen   bestskill2(Missile, Bows)
items    0x6c                            # Trueflight
option 0                                   # shoot for the popinjay
  who    bestskill2(Missile, Bows)
  chance clamp(5, 85, max(skill(who, Missile), skill(who, Bows)) + attr(who, Per) / 3 - 20)
  ok     chosen who; if skill(who, Missile) >= skill(who, Bows) then practice who, Missile, 1, 10, 0 else practice who, Bows, 1, 10, 0
         goto king
  fail   chosen who; text 1; hours 2
         if skill(who, Missile) >= skill(who, Bows) then practice who, Missile, 0, 10, 0 else practice who, Bows, 0, 10, 0
         exit
option 1 potion                            # dress the shot with a potion first
  potion 0x6c, clamp(0, 99, max(skill(who, Missile), skill(who, Bows)) + attr(who, Per) / 3 + q / 2 - 10)
  ok     chosen who; goto king
  fail   chosen who; text 1; hours 2; exit
option 2                                   # watch from the beer-bench
  chance none
  ok     text 2; attr party, End, 2; hours 2; exit
option 3                                   # walk on
  chance none
  ok     text 3; exit

[king]
block    4
option 0                                   # stand the guild its beer like a king
  need   money >= money1
  chance none
  ok     text 5; pay money1; fame 1; reputation 1, 3; hours 2; exit
option 1                                   # plead an empty purse, royally
  who    bestattr(Cha)
  chance clamp(5, 90, (2 * attr(who, Cha) + skill(who, SpeakCommon)) / 3 - 10)
  ok     chosen who; text 6; fame 1; reputation 1, 2; practice who, SpeakCommon, 1, 10, 0; hours 2; exit
  fail   chosen who; text 7; reputation -1, -2; practice who, SpeakCommon, 0, 10, 0; exit
option 2                                   # give the chain to the town's own champion
  chance none
  ok     text 8; reputation 1, 2; exit
option 3                                   # make off with the silver chain
  who    bestskill(Stealth)
  chance clamp(5, 80, skill(who, Stealth) + attr(who, Agl) / 3 - 15)
  ok     chosen who; text 9; gain money2; attr party, DF, -3; practice who, Stealth, 1, 10, 0; exit
  fail   chosen who; text 10; injure who, 1, 1, 5; reputation -3, -6; attr party, DF, -2; exit
