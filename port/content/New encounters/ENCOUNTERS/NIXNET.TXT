# The net in the rudder: a poacher's net fouls the steering oar, and the Wassermann is caught in it.
name     The net in the rudder
card     $NvNix00
picture  ARIVE
where    boat
when     any
weight   1

[start]
block    0
chosen   bestskill(Edged)
saints   0x63, 0x16, 0x6f                # Nicholas, Christopher, Raphael
option 0                                   # cut the oar free (Edged)
  who    bestskill(Edged)
  chance clamp(5, 90, skill(who, Edged) + attr(who, Agl) / 3 - 5)
  ok     chosen who; text 1; attr party, End, 2; practice who, Edged, 1, 10, 0; exit
  fail   chosen who; text 2; attr who, End, -4; practice who, Edged, 0, 10, 0; hours 3; exit
option 1                                   # haul the net aboard (the party's Strength)
  who    party
  chance clamp(10, 90, avgattr(Str) + size * 5)
  ok     goto nix
  fail   text 4; attr party, End, -2; exit
option 2 saint
  ok     text 5; reputation 1, 2; exit
  fail   text 6
option 3                                   # let the boatmen pole her clear
  chance none
  ok     text 7; hours 3; attr party, End, -2; exit

[nix]
block    3
option 0                                   # cut him loose and let him go
  chance none
  ok     money1 random(48) + 24; text 8; gain money1; exit
option 1                                   # hold him till he swears to speed you to the landing (Streetwise)
  who    bestskill(Streetwise)
  chance clamp(5, 90, skill(who, Streetwise) + attr(who, Int) / 3 - 10)
  ok     chosen who; text 9; attr party, End, 3; practice who, Streetwise, 1, 10, 0; attr party, DF, -2; exit
  fail   chosen who; text 10; attr who, End, -5; practice who, Streetwise, 0, 10, 0; exit
option 2                                   # bind him in God's name (Religion)
  who    bestskill(Religion)
  chance clamp(5, 90, skill(who, Religion) + attr(who, Int) / 3 - 10)
  ok     chosen who; text 11; practice who, Religion, 1, 10, 0; attr party, DF, 3; reputation 1, 3; exit
  fail   chosen who; text 12; practice who, Religion, 0, 10, 0; attr who, End, -3; exit
