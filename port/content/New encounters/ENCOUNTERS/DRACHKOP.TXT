# The dragon's skull: quarrymen break into a cave of great bones, and down tools.
name     The dragon's skull
card     $NhDrk00
picture  MS016
where    wild
terrain  hills mountains
when     day
weight   1

[start]
do       money1 24 + size * 6
block    0
chosen   bestskillsum(Latin, ReadWrite)
option 0                                   # pronounce learnedly on it (Latin, ReadWrite)
  who    bestskillsum(Latin, ReadWrite)
  chance clamp(5, 90, (skill(who, Latin) + skill(who, ReadWrite)) / 2 + attr(who, Int) / 3 - 10)
  ok     chosen who; text 1; gain money1; practice who, Latin, 1, 10, 0; practice who, ReadWrite, 1, 5, -1; exit
  fail   chosen who; text 2; practice who, Latin, 0, 10, 0; reputation -1, -2; exit
option 1                                   # bless the cave and the quarry (Religion)
  who    bestskill(Religion)
  chance clamp(10, 90, skill(who, Religion) + attr(who, Cha) / 3 - 5)
  ok     chosen who; text 3; hours 1; attr party, DF, 2; practice who, Religion, 1, 10, 0; reputation 1, 2; exit
  fail   chosen who; text 4; hours 1; practice who, Religion, 0, 10, 0; exit
option 2                                   # swear it is a dragon and sell them charms (Streetwise)
  who    bestskill(Streetwise)
  chance clamp(5, 90, skill(who, Streetwise) + attr(who, Cha) / 3 - 10)
  ok     chosen who; money1 random(48) + 36; text 5; gain money1; attr party, DF, -3; practice who, Streetwise, 1, 10, 0; exit
  fail   chosen who; text 6; reputation -1, -3; practice who, Streetwise, 0, 10, 0; exit
option 3                                   # go on
  chance none
  ok     text 7; exit
