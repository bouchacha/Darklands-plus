# The bee-tree: a hollow oak full of wild honey, and its bees.
name     The bee-tree
card     $NwBee00
picture  XTREE
where    wild
terrain  woods forest farmland
when     day
weight   1
only     month >= 5 and month <= 9

[start]
block    0
option 0                                   # smoke them out (woodcraft)
  who    bestskill(Woodwise)
  chance clamp(10, 90, skill(who, Woodwise) + attr(who, Per) / 3)
  ok     chosen who; text 1; hours 1; attr party, End, 3 + random(3); practice who, Woodwise, 1, 10, 0; exit
  fail   chosen who; text 2; hours 1; attr party, End, -(2 + random(3)); practice who, Woodwise, 0, 10, 0; exit
option 1                                   # climb up (agility)
  who    bestattr(Agl)
  chance clamp(5, 80, attr(who, Agl) + attr(who, Str) / 3 - 15)
  ok     chosen who; text 3; hours 1; attr party, End, 2 + random(3); attr who, End, -2; exit
  fail   chosen who; text 4; injure who, 1, 1, 10; exit
option 2                                   # wait for dusk and take it quietly (stealth)
  who    party
  chance clamp(10, 90, avgskill(Stealth) + avgagility / 2)
  ok     text 5; until 20; attr party, End, 3 + random(3); practice party, Stealth, 1, 10, 0; exit
  fail   text 6; until 20; attr party, End, -(2 + random(3)); practice party, Stealth, 0, 10, 0; exit
option 3                                   # leave them
  chance none
  ok     text 7; exit
