# The grave robbers: lanterns in a lonely churchyard at night.
name     The grave robbers
card     $NwGrv00
picture  XGRV2
terrain  farmland plains woods road
when     night
weight   1

[start]
block    0
saints   0x60, 0x55, 0x39, -1            # Michael, Lazarus, Gertrude of Nivelles
option 0                                   # fall upon them
  chance combat
  ok     goto fight
option 1                                   # bellow at them like the wrath of God
  who    bestattr(Cha)
  chance clamp(5, 85, (attr(who, Cha) + attr(who, Str)) / 2 + min(20, fame / 10) - 10)
  ok     chosen who; text 1; attr party, DF, 2; exit
  fail   chosen who; text 2; goto fight
option 2 saint                             # call on a saint to guard the dead
  ok     text 3; attr party, DF, 3; exit
  fail   text 4
option 3                                   # wait, and help yourselves when they have gone
  chance none
  ok     money1 random(48) + 24; gain money1; text 5; attr party, DF, -15; skill party, Virtue, -2; until 5; exit
option 4                                   # leave the dead to their fate
  chance none
  ok     text 6; exit

[fight]
do       battle Wilds, Thug, lore / 4 + 1, clamp(size, 6, random(4) + lore / 3 + 1), Henchman, lore % 3 + 1, 1
         if result == Won then { text 7; hours 1; attr party, DF, 3; reputation 1, 2; fame 1; exit }
         if result == Fled then { rejoin; text 8; until 5; exit }
         loseitems party; text 9; until 5; exit
