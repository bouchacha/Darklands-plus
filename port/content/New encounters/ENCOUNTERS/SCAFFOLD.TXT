# The falling beam: builders' scaffolding gives way above the street.
name     The falling beam
card     $NsScf00
picture  XARMOR
where    street
when     day
weight   1

[start]
do       money1 random(24) + 24
block    0
chosen   bestattr(Agl)
option 0                                   # leap clear
  who    slowest
  chance clamp(10, 95, minagility + 30)
  ok     text 1; gain money1; exit
  fail   chosen slowest; text 2; injure slowest, 1, 1, 10; gain money1; exit
option 1                                   # brace yourselves and catch the beam
  who    bestattr(Str)
  chance clamp(5, 80, attr(who, Str) + attr(who, End) / 2 - 20)
  ok     chosen who; text 3; attr who, End, -3; fame 1; reputation 1, 2; exit
  fail   chosen who; text 4; injure who, 1, 2, 10; reputation 0, 2; exit
option 2                                   # snatch the child out of its way
  who    bestattr(Agl)
  chance clamp(10, 90, attr(who, Agl) + attr(who, Per) / 3 - 5)
  ok     chosen who; text 5; attr party, DF, 2; reputation 1, 3; exit
  fail   chosen who; text 6; injure who, 1, 1, 10; attr party, DF, 1; reputation 1, 2; exit
