# The crier's proclamation: the town crier describes a thief, and every head in the square turns to you.
name     The crier's proclamation
card     $NdCry00
picture  POST-UP
where    street
when     day
weight   1

[start]
do       money1 240
block    0
chosen   bestattr(Str)
option 0                                   # ask the crier to read on
  chance none
  ok     text 1; goto spot
option 1                                   # face the crowd down
  who    bestattr(Cha)
  chance clamp(5, 90, (2 * attr(who, Cha) + skill(who, SpeakCommon)) / 3 + min(15, fame / 10) - 5)
  ok     chosen who; text 2; practice who, SpeakCommon, 1, 10, 0; goto spot
  fail   chosen who; text 3; reputation -1, -2; practice who, SpeakCommon, 0, 10, 0; hours 3; exit
option 2                                   # leave the square quickly
  chance none
  ok     text 4; reputation -1, -2; exit

[spot]
do       chosen2 spotter
block    5
option 0                                   # cut him off at the end of the lane
  who    bestattrskill(Agl, Streetwise)
  chance clamp(5, 90, attr(who, Agl) + skill(who, Streetwise) / 2 - 10)
  ok     chosen who; text 6; gain money1; reputation 1, 2; practice who, Streetwise, 1, 10, 0; hours 1; exit
  fail   chosen who; text 7; practice who, Streetwise, 0, 10, 0; exit
option 1                                   # follow him quietly to his lair
  who    bestskill(Stealth)
  chance clamp(5, 90, skill(who, Stealth) + attr(who, Per) / 3 - 10)
  ok     chosen who; text 8; gain money1; reputation 1, 3; practice who, Stealth, 1, 10, 0; hours 2; exit
  fail   chosen who; text 9; practice who, Stealth, 0, 10, 0; hours 1; exit
option 2                                   # point him out to the watch
  chance none
  ok     if roll(40) then { money1 60; text 10; gain money1; reputation 1, 1; hours 1 } else { text 11; hours 1 }
         exit
