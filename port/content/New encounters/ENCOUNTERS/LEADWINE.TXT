  who    bestskill(Alchemy)

  who    bestskill(Artifice)
  chance clamp(5, 90, skill(who, Artifice) + attr(who, Agl) / 3 - 10)
  ok     chosen who; text 4; practice who, Artifice, 1, 10, 0; goto lead
  fail   chosen who; text 5; hours 1; practice who, Artifice, 0, 10, 0; disable 2
option 3                                   # see to the carter (Healing)
  who    bestskill(Healing)
  chance clamp(10, 90, skill(who, Healing) + attr(who, Int) / 3 - 5)
  ok     chosen who; text 6; attr party, DF, 1; practice who, Healing, 1, 10, 0; goto lead
  fail   chosen who; text 7; hours 1; practice who, Healing, 0, 10, 0; disable 3
option 4                                   # leave the wine and go out
  chance none
  ok     text 8; exit
