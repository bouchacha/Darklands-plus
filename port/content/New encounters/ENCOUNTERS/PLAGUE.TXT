# The village behind the cross: a village whose doors are marked with plague crosses.
name     The village behind the cross
card     $NwPlq00
picture  XGRV1
where    wild
terrain  farmland plains woods road
when     day
weight   1

[start]
do       money1 size * 12
block    0
chosen   bestskill(Healing)
saints   0x73, 0x75, 0x55, 0x6f          # Roch, Sebastian, Lazarus, Raphael
option 0                                   # go in and tend the sick
  who    bestskill(Healing)
  chance clamp(5, 85, skill(who, Healing) + attr(who, Int) / 4 - 10)
  ok     chosen who; text 1; days 2; practice who, Healing, 1, 10, 0; attr party, DF, 3; reputation 1, 3
         if roll(20) then { text 3; attr who, End, -(3 + random(4)) }
         exit
  fail   chosen who; text 2; days 2; practice who, Healing, 0, 10, 0; attr party, DF, 1
         attr who, End, -(4 + random(4)); attr who, Str, -1; exit
option 1 saint                             # pray to a saint for the village
  ok     text 4; hours 2; reputation 1, 2; exit
  fail   text 5
option 2                                   # leave alms in the vinegar bowl
  need   money >= money1
  chance none
  ok     text 6; pay money1; attr party, DF, 1 + random(2); exit
option 3                                   # go around across the fields
  chance none
  ok     text 7; hours 4 + random(4); exit
