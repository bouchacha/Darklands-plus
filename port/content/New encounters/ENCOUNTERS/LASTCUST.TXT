[start]

option 1                                   # carry him out for the keeper
  chance none
  ok     text 1; attr party, End, 2; exit
option 2                                   # leave the keeper to his troubles
  chance none
  ok     text 2; exit

[house]

  who    bestattrskill(Cha, Woodwise)

  ok     chosen who; text 4; additem leader, 0x3a; reputation 1, 2; practice who, Woodwise, 1, 10, 0; exit

  who    bestattrsum(Agl, Per)
  chance clamp(5, 75, (attr(who, Agl) + attr(who, Per)) / 2 - 15)
  ok     chosen who; money1 24 + random(36); text 6; gain money1; exit

  ok     text 8; reputation 1, 2; exit
