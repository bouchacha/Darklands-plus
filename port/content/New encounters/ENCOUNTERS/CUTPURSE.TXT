# The cutpurse: a hand at a purse in the market crowd.
name     The cutpurse
card     $NsCut00
picture  MAIN-ST
where    street
when     day
weight   2
only     money >= 24

[start]
do       money1 min(money, clamp(12, 240, money / 8))
block    0
chosen   bestattr(Per)
option 0                                   # go after him and seize him
  who    bestattrsum(Agl, Per)
  chance clamp(5, 90, (attr(who, Agl) + attr(who, Per)) * 2 / 3 + skill(who, Streetwise) / 4 - 5)
  ok     chosen who; practice who, Streetwise, 1, 10, 0; goto caught
  fail   chosen who; text 2; pay money1; practice who, Streetwise, 0, 10, 0; exit
option 1                                   # raise the hue and cry
  who    bestattr(Cha)
  chance clamp(5, 90, 30 + standing / 3 + attr(who, Cha) / 2)
  ok     chosen who; text 3; reputation 1, 2; hours 1; exit
  fail   chosen who; text 4; pay money1; reputation -1, -2; exit
option 2                                   # let him go
  chance none
  ok     text 5; pay money1; exit

[caught]
block    1
option 0                                   # hand him over to the watch
  chance none
  ok     text 6; reputation 1, 3; hours 1; exit
option 1                                   # box his ears and let him run
  chance none
  ok     text 7; attr party, DF, 2; practice party, Virtue, 1, 10, 0; exit
option 2                                   # keep the other two purses
  chance none
  ok     money1 random(48) + 18; text 8; gain money1; attr party, DF, -3; exit
