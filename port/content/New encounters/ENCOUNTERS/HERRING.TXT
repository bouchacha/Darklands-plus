# The herring barrel: a barrel falls from the porters' ropes and the merchant blames you; the rope was cut, and the herring under the top layer stink.
name     The herring barrel
card     $NdHer00
picture  XSIDE
where    street
when     day
weight   1

[start]
do       money1 clamp(60, 240, money / 8); money2 money1 / 2
block    0
chosen   bestattr(Str)
option 0                                   # look at the broken rope and the barrel
  who    bestattrskill(Per, Streetwise)
  chance clamp(5, 90, attr(who, Per) + skill(who, Streetwise) / 2 - 10)
  ok     chosen who; practice who, Streetwise, 1, 10, 0; goto found
  fail   chosen who; money2 min(money, money2); text 1; pay money2; practice who, Streetwise, 0, 10, 0; exit
option 1                                   # argue it out before the crowd
  who    bestattr(Cha)
  chance clamp(5, 90, (2 * attr(who, Cha) + skill(who, SpeakCommon)) / 3 + standing / 10 - 5)
  ok     chosen who; text 2; reputation 1, 2; practice who, SpeakCommon, 1, 10, 0; exit
  fail   chosen who; money2 min(money, money2); text 3; pay money2; practice who, SpeakCommon, 0, 10, 0; exit
option 2                                   # pay what he asks
  need   money >= money1
  chance none
  ok     text 4; pay money1; reputation 1, 2; exit
option 3                                   # shake him off and walk on
  chance none
  ok     text 5; reputation -1, -2; exit

[found]
block    6
option 0                                   # call the beadle over
  chance none
  ok     text 7; reputation 1, 3; exit
option 1                                   # take what he asked of you and hold your tongue
  chance none
  ok     text 8; gain money1; attr party, DF, -2; exit
