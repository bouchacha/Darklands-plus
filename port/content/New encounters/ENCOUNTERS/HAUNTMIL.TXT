# The haunted mill: a mill that grinds by night with no miller.
name     The haunted mill
card     $NwHmi00
picture  XMS056
where    wild
terrain  farmland woods forest plains
when     night
weight   1

[start]
block    0
chosen   bestskill(Religion)
option 0                                   # cast the spirit out
  who    bestskill(Religion)
  chance clamp(5, 90, skill(who, Religion) + attr(who, DF) / 3 + skill(who, Latin) / 4 - 15)
  ok     chosen who; text 1; practice who, Religion, 1, 10, 0; attr party, DF, 3; reputation 1, 2; hours 1; exit
  fail   chosen who; text 2; practice who, Religion, 0, 10, 0; injure who, 1, 1, 5; attr who, DF, -3; hours 1; exit
option 1                                   # keep watch in the mill
  chance assured
  ok     text 3; if hour >= 12 then until 0 else hours 1
         goto hoard
option 2                                   # hurry on past
  chance none
  ok     text 4; exit

[hoard]
block    5
option 0                                   # keep the coins
  chance none
  ok     money1 random(60) + 36; text 6; gain money1; attr party, DF, -5; until 5; exit
option 1                                   # bring them to the priest for the poor
  chance none
  ok     text 7; until 5; hours 2; attr party, DF, 4; practice party, Virtue, 1, 10, 0; exit
