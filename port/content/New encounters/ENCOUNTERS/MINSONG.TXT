# The minstrel's song: a village minstrel sings of your deeds, and in his song you are the villains.
name     The minstrel's song
card     $NrMin00
picture  XMOBGLAD
where    wild
terrain  road farmland plains
when     day
weight   1
only     fame >= 10                        # a song needs a name to hang on

[start]
do       money1 60
block    0
chosen   bestattr(Cha)
option 0                                   # tell the true tale (SpeakCommon)
  who    bestattr(Cha)
  chance clamp(5, 90, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5 + min(20, fame / 10) - 10)
  ok     chosen who; text 1; fame 1; hours 1; practice who, SpeakCommon, 1, 10, 0; exit
  fail   chosen who; text 2; hours 1; attr who, End, -2; practice who, SpeakCommon, 0, 10, 0; exit
option 1                                   # play him down (a harp or a flute)
  need   has(party, 0x41) or has(party, 0x42)
  who    bestattr(Cha)
  chance clamp(5, 90, attr(who, Cha) + attr(who, Agl) / 3 + has(party, 0x41) * 10 - 10)
  ok     chosen who; text 3; fame 1; reputation 1, 2; hours 2; exit
  fail   chosen who; text 4; hours 1; attr who, End, -2; exit
option 2                                   # pay him to change the ending
  need   money >= money1
  chance none
  ok     text 5; pay money1; fame 1; exit
option 3                                   # pull him off his barrel
  chance none
  ok     text 6; reputation -1, -3; exit
option 4                                   # laugh loudest of all
  chance none
  ok     text 7; attr party, DF, 1; practice party, Virtue, 1, 10, 0; exit
