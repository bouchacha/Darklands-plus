# The feud: a lord's riders burn his enemy's manor and village, as feuds are fought.
name     The feud
card     $NwFeu00
picture  XFIRE2
terrain  farmland plains woods road
when     any
weight   1

[start]
block    0
chosen   bestattr(Cha)
saints   0x33, 0x01, 0x37, 0x5d          # Florian, Agatha, George, Martin of Tours
option 0                                   # drive the raiders off
  chance combat
  ok     goto fight
option 1                                   # help the villagers save what they can
  who    party
  chance clamp(10, 90, avgskill(Stealth) + avgagility * 2 / 3)
  ok     text 1; hours 3; attr party, DF, 2; reputation 1, 2; exit
  fail   text 2; goto fight
option 2 saint
  ok     text 3; attr party, DF, 1; reputation 1, 3; exit
  fail   text 4
option 3                                   # speak with the knight who leads them
  who    bestattr(Cha)
  chance clamp(1, 95, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5 - 10)
  ok     chosen who; text 5; practice who, SpeakCommon, 1, 10, 0; attr who, DF, 2; reputation 1, 2; hours 1; exit
  fail   chosen who; text 6; practice who, SpeakCommon, 0, 10, 0; goto fight
option 4                                   # join in the plunder
  ok     money1 random(48) + 24; gain money1; text 7; attr party, DF, -8; skill party, Virtue, -1; reputation -1, -3; hours 2; exit
option 5                                   # go around
  ok     text 11; exit

[fight]
do       battle Clearing, Trooper, lore / 4 + 1, random(5) + 3, Knight, lore % 3 + 1, 1
         if result == Won then { text 8; hours 2; attr party, DF, 2; reputation 1, 3; fame 1; exit }
         if result == Fled then { rejoin; text 9; exit }
         loseitems party; text 10; exit
