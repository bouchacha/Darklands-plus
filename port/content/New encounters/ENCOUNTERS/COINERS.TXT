# The coiners' cave: hammering in a cleft of the hills, and groschen that no mint ever struck.
name     The coiners' cave
card     $NhCoi00
picture  MS015
where    wild
terrain  hills mountains pinewood
when     day
weight   1

[start]
block    0
chosen   bestskill(Stealth)
option 0                                   # creep close, mark their faces and fetch the Vogt (Stealth)
  who    bestskill(Stealth)
  chance clamp(5, 90, skill(who, Stealth) + attr(who, Agl) / 3 - 5)
  ok     chosen who; money1 random(60) + 60; text 1; hours 6; gain money1; reputation 1, 3; practice who, Stealth, 1, 10, 0; exit
  fail   chosen who; text 2; practice who, Stealth, 0, 10, 0; goto fight
option 1                                   # walk in as fellow rogues and bargain for a share (Streetwise)
  who    bestskill(Streetwise)
  chance clamp(5, 90, skill(who, Streetwise) + attr(who, Cha) / 3 - 5)
  ok     chosen who; money1 random(48) + 48; text 3; gain money1; attr party, DF, -3; practice who, Streetwise, 1, 10, 0; exit
  fail   chosen who; text 4; attr party, DF, -2; practice who, Streetwise, 0, 10, 0; exit
option 2                                   # rush them at their forge
  chance combat
  ok     text 5; goto fight
option 3                                   # back away quietly
  chance none
  ok     text 6; exit

[fight]
do       battle Clearing, Thug, lore / 4 + 1, random(3) + 3, BrigandSergeant, lore % 3 + 1, 1
         if result == Won then { money1 random(60) + 60; text 7; hours 6; gain money1; fame 1; reputation 1, 3; exit }
         if result == Fled then { rejoin; text 8; hours 2; exit }
         if result == Surrendered then { text 10; keepitems party, 50; hours 2; exit }
         text 9; keepitems party, 50; hours 3; exit
