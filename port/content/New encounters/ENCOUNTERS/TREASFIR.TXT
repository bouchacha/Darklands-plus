# The treasure fire: a blue flame in the night marks buried treasure, which must be dug in silence; and who buried it.
name     The treasure fire
card     $NkTrs00
picture  XMS028
where    camp
when     any
weight   1
only     size >= 2
vars     read                                  # 1: the Latin on the lid was read

[start]
block    0
chosen   worstattr(Int)                        # the one least likely to hold $his tongue
saints   0x60, 0x09, -1, -1                    # Michael, Anthony (who fled the Devil's gold in the desert)
option 0                                   # dig in silence (the party's Stealth and the chatterbox's wits)
  who    party
  chance clamp(5, 85, minattr(Int) + avgskill(Stealth) / 2 + 10)
  ok     text 2; hours 2; goto found
  fail   text 1; hours 3; attr party, End, -3; exit
option 1 saint                             # pray before you put a spade in the ground
  ok     text 3; hours 2; goto found
  fail   text 4
option 2                                   # leave the Devil's lantern alone
  chance none
  ok     text 5; exit

[found]
picture  XTRSCHST
do       money1 random(96) + 48
block    6
option 0                                   # puzzle out the Latin on the lid (Latin)
  who    bestskill(Latin)
  chance clamp(5, 90, skill(who, Latin) + attr(who, Int) / 3 - 5)
  ok     chosen who; set read = 1; text 7; practice who, Latin, 1, 10, 0; disable 0
  fail   chosen who; text 8; practice who, Latin, 0, 10, 0; disable 0
option 1                                   # take the silver and give the bones a Christian burial
  only   read == 0
  chance none
  ok     text 9; gain money1; hours 1; attr party, DF, 1; exit
option 2                                   # take the silver and go
  chance none
  ok     text 10; gain money1; attr party, DF, -3 - read * 2; exit
option 3                                   # carry it all back to the church it came from
  only   read == 1
  chance none
  ok     text 11; hours 4; attr party, DF, 3; reputation 1, 3; exit
