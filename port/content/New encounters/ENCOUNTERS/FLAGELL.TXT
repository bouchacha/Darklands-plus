# The flagellants: a procession of penitents scourging themselves along the road.
name     The flagellants
card     $NwFlg00
picture  XMOBGLAD
terrain  road farmland plains
when     day
weight   1

[start]
do       money1 size * 12
block    0
chosen   bestskill(Religion)
option 0                                   # join their penance
  chance assured
  ok     text 1; hours 24
         attr party, End, -(3 + random(4)); attr party, Str, -1
         attr party, DF, skill(who, Religion) / 4 + 5; practice party, Virtue, 1, 15, 0; exit
option 1                                   # give alms
  need   money >= money1
  chance none
  ok     pay money1; text 2; attr party, DF, 2 + skill(who, Religion) / 8; practice party, Virtue, 1, 5, -1; exit
option 2                                   # dispute with their preacher
  who    bestskill(Religion)
  chance clamp(5, 90, skill(who, Religion) + skill(who, Latin) / 2 + attr(who, Cha) / 3 - 20)
  ok     text 3; practice who, Religion, 1, 15, 0; reputation 1, 3; hours 1; exit
  fail   text 4; practice who, Religion, 0, 10, 0; goto mob
option 3                                   # stand aside
  chance none
  ok     text 5; hours 1; exit

[mob]
do       battle Clearing, Villager, lore / 4 + 1, random(5) + 4, Friar, lore % 3 + 1, 1
         if result == Won goto won
         if result == Fled goto fled
         text 8; until 18; attr party, End, -3; exit
[won]
do       text 6; hours 1; attr party, DF, -10; exit
[fled]
do       rejoin; text 7; until 18; exit
