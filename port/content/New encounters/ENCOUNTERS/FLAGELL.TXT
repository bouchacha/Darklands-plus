# The flagellants: a procession of penitents scourging themselves along the road.
name     The flagellants
card     $NwFlg00
picture  XMOBGLAD
terrain  road farmland plains
when     day
weight   1

[start]
do       money1 size * 12
block    0
chosen   bestskill(Religion)
option 0                                   # join their penance
  chance assured
  ok     text 1; until 18
         attr party, End, -(3 + random(4)); attr party, Str, -1
         attr party, DF, 2 + skill(who, Religion) / 10; practice party, Virtue, 1, 10, 0; exit
option 1                                   # give alms
  need   money >= money1
  chance none
  ok     pay money1; text 2; attr party, DF, 1 + skill(who, Religion) / 10; practice party, Virtue, 1, 5, -1; exit
option 2                                   # dispute with their preacher
  who    bestskill(Religion)
  chance clamp(5, 90, skill(who, Religion) + skill(who, Latin) / 2 + attr(who, Cha) / 3 - 20)
  ok     chosen who; text 3; practice who, Religion, 1, 10, 0; reputation 1, 3; hours 1; exit
  fail   chosen who; text 4; practice who, Religion, 0, 10, 0; goto mob
option 3                                   # stand aside
  chance none
  ok     text 5; hours 1; exit

[mob]
do       battle Clearing, Villager, lore / 4 + 1, clamp(size, 7, random(5) + lore / 3 + 1), Friar, lore % 3 + 1, 1
         if result == Won goto won
         if result == Fled goto fled
         text 8; until 18; attr party, End, -3; exit
[won]
do       text 6; hours 1; attr party, DF, -4; reputation -1, -3; exit
[fled]
do       rejoin; text 7; until 18; exit
