# A body in the gutter: a man stabbed in a dark lane, and the bier-right that names his killer.
name     A body in the gutter
card     $NnGut00
picture  DEADROB
where    street
when     night
weight   1
vars     clue, called

[start]
block    0
chosen   bestattrskill(Per, Healing)
option 0                                   # look at the wound (Healing)
  who    bestattrskill(Per, Healing)
  chance clamp(5, 90, skill(who, Healing) + attr(who, Per) / 2 - 10)
  ok     chosen who; set clue = 1; text 1; practice who, Healing, 1, 10, 0; goto watch
  fail   chosen who; text 2; practice who, Healing, 0, 10, 0; goto watch
option 1                                   # raise the hue and cry
  chance none
  ok     set called = 1; goto watch
option 2                                   # go through his purse (the watch may come)
  chance none
  ok     money1 random(30) + 18; if roll(50) goto caught
         text 3; gain money1; attr party, DF, -3; exit
option 3                                   # walk on
  chance none
  ok     text 4; exit

[caught]
do       money1 min(money, 48 + size * 24); text 5; pay money1; until 8; reputation -2, -4; attr party, DF, -2; exit

[watch]
block    6
option 0                                   # speak for yourselves now
  who    bestattr(Cha)
  chance clamp(5, 90, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5 + clue * 25 + called * 10 - 10)
  ok     chosen who; text 7; reputation 1, 3; if clue then fame 1
         practice who, SpeakCommon, 1, 10, 0; hours 1; exit
  fail   chosen who; text 8; practice who, SpeakCommon, 0, 10, 0; goto bier
option 1                                   # submit to the bier-right in the morning
  chance none
  ok     goto bier
option 2                                   # slip away into the dark
  who    party
  chance clamp(5, 90, avgskill(Stealth) + avgagility * 2 / 3 - 10)
  ok     text 9; reputation -1, -3; practice party, Stealth, 1, 10, 0; hours 1; exit
  fail   text 10; practice party, Stealth, 0, 10, 0; goto bier

[bier]
picture  MS041
do       until 8; text 11; reputation 1, 3; exit
