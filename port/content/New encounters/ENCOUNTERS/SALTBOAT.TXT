# The salt smugglers: at slaughtering time, men land salt from a flat boat in the reeds by night against the lord's salt-house, and their leader wears a tonsure.
name     The salt smugglers
card     $NmSlt00
picture  XHIDDEN
where    wild
terrain  marsh woods road
when     night
weight   1
only     month >= 10 and month <= 12           # slaughtering time, when the winter pork is salted

[start]
do       money1 size * 24
block    0
option 0                                   # take his coins and see nothing
  chance none
  ok     text 1; gain money1; attr party, DF, -1; exit
option 1                                   # help carry the sacks before the riders pass (Stealth)
  who    party
  chance clamp(5, 90, avgskill(Stealth) + avgagility * 2 / 3)
  ok     text 2; until 4; attr party, DF, 3; practice party, Stealth, 1, 10, 0; exit
  fail   text 3; until 4; reputation -1, -3; practice party, Stealth, 0, 10, 0; exit
option 2                                   # suggest that silence is worth more (Streetwise)
  who    bestskill(Streetwise)
  chance clamp(5, 90, skill(who, Streetwise) + attr(who, Cha) / 3 - 10)
  ok     chosen who; money2 money1 * 3; text 4; gain money2; attr party, DF, -4; practice who, Streetwise, 1, 10, 0; exit
  fail   chosen who; text 5; practice who, Streetwise, 0, 10, 0; exit
option 3                                   # walk on
  chance none
  ok     text 6; exit
