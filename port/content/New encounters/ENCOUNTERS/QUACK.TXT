# The quack's cart: a doctor with a painted cart sells a cure for everything.
name     The quack's cart
card     $NsQck00
picture  BADDALC
where    street
when     day
weight   1

[start]
do       money1 clamp(24, 72, money / 20)
block    0
chosen   bestskill2(Healing, Alchemy)
option 0                                   # buy a flask (one in four is a real healing draught)
  need   money >= money1
  chance none
  ok     if roll(25) then { text 1; pay money1; additem leader, 0x73, 10 + random(16) } else { text 2; pay money1 }
         exit
option 1                                   # show the crowd what he is
  who    bestskill2(Healing, Alchemy)
  chance clamp(5, 90, max(skill(who, Healing), skill(who, Alchemy)) + attr(who, Int) / 3 - 10)
  ok     chosen who; if skill(who, Healing) >= skill(who, Alchemy) then practice who, Healing, 1, 10, 0 else practice who, Alchemy, 1, 10, 0
         goto unmasked
  fail   chosen who; text 3; reputation -1, -2; disable 1
         if skill(who, Healing) >= skill(who, Alchemy) then practice who, Healing, 0, 10, 0 else practice who, Alchemy, 0, 10, 0
option 2                                   # leave him to his customers
  chance none
  ok     text 4; exit

[unmasked]
block    5
option 0                                   # tell the crowd
  chance none
  ok     text 6; reputation 1, 3; hours 1; exit
option 1                                   # take his quarter and hold your tongue
  chance none
  ok     money1 random(36) + 24; text 7; gain money1; attr party, DF, -3; exit
