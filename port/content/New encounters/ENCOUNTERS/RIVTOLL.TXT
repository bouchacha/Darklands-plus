# The toll castle: a chain across the river, and a lord who wants paying.
name     The toll castle
card     $NbTol00
picture  CASTLE1
where    boat
when     any
weight   1

[start]
do       money1 clamp(24, 360, money / 10 + size * 12)
block    0
chosen   bestattr(Cha)
option 0                                   # pay the toll
  need   money >= money1
  chance none
  ok     text 1; pay money1; exit
option 1                                   # claim to be pilgrims (Speak Common, Charisma)
  who    bestattr(Cha)
  chance clamp(1, 95, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5 - 5)
  ok     chosen who; text 2; practice who, SpeakCommon, 1, 10, 0; attr party, DF, -1; exit
  fail   chosen who; money1 money1 * 2; text 3; practice who, SpeakCommon, 0, 10, 0; disable 1
option 2                                   # bargain for the whole boat, as the captain asks (Streetwise)
  who    bestskill(Streetwise)
  chance clamp(5, 90, skill(who, Streetwise) + attr(who, Cha) / 3 - 5)
  ok     chosen who; money1 size * 24; text 4; gain money1; practice who, Streetwise, 1, 10, 0; hours 1; exit
  fail   chosen who; money1 min(money, money1); text 5; pay money1; practice who, Streetwise, 0, 10, 0; hours 12; exit
