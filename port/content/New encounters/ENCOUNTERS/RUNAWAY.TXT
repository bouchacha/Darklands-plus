# The runaway serf: a bondman fleeing his lord begs you to hide him; the lord's riders follow.
name     The runaway serf
card     $NwRun00
picture  MS017
where    wild
terrain  farmland plains woods road
when     day
weight   1

[start]
block    0
option 0                                   # hide him and see him to the town
  who    party
  chance clamp(5, 90, avgskill(Stealth) + avgagility / 2)
  ok     text 1; hours 4; practice party, Stealth, 1, 10, 0; practice party, Virtue, 1, 10, 0; attr party, DF, 2; tocity
  fail   text 2; practice party, Stealth, 0, 10, 0; goto riders
option 1                                   # outride the lord's men
  need   mounted
  who    party
  chance clamp(5, 90, minskill(Riding) * 2 + 10)
  ok     text 3; hours 3; practice party, Riding, 1, 10, 0; practice party, Virtue, 1, 10, 0; attr party, DF, 2; tocity
  fail   text 4; practice party, Riding, 0, 10, 0; goto riders
option 2                                   # hand him over to the riders
  chance none
  ok     money1 48; text 5; gain money1; attr party, DF, -2; hours 1; exit
option 3                                   # want no part of it
  chance none
  ok     text 6; exit

[riders]
picture  ROBMEN2
do       money1 min(money, clamp(24, 240, money / 8))
block    7
option 0                                   # give him up and pay the fine
  chance none
  ok     text 8; pay money1; exit
option 1                                   # fight for him
  chance combat
  ok     goto fight

[fight]
do       battle Wilds, Trooper, lore / 4 + 1, random(3) + 3, Sergeant, lore % 3 + 1, 1
         if result == Won then { text 9; hours 3; attr party, DF, 2; practice party, Virtue, 1, 10, 0; reputation -1, -3; tocity }
         if result == Fled then { rejoin; text 10; hours 2; exit }
         money1 min(money, money1 * 2); text 11; pay money1; hours 2; exit
