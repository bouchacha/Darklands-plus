# A knight's challenge: a young knight holds a bridge for his lady's honour against all comers.
name     A knight's challenge
card     $NwChl00
picture  KNGHTCLO
where    wild
terrain  road plains farmland woods
when     day
weight   1
vars     mocked

[start]
do       set mocked = 0
block    0
chosen   bestattr(Cha)
option 0                                   # accept his challenge
  chance combat
  ok     text 1; goto fight
option 1                                   # decline with courtesy
  who    bestattr(Cha)
  chance clamp(5, 90, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5 + fame / 20)
  ok     chosen who; text 2; practice who, SpeakCommon, 1, 10, 0; hours 1; attr party, End, 2 + random(2); exit
  fail   chosen who; text 3; practice who, SpeakCommon, 0, 10, 0; hours 4 + random(3); exit
option 2                                   # mock him and his lady
  chance combat
  ok     chosen bestattr(Cha); set mocked = 1; text 4; goto fight
option 3                                   # cross by the ford downstream
  chance none
  ok     text 5; hours 3 + random(3); exit

[fight]
do       if mocked then battle Clearing, Knight, lore / 5, 1 else battle Clearing, Knight, lore / 3 + 2, 1
         if result == Won and mocked then { money1 random(48) + 36; text 6; gain money1; reputation -1, -3; exit }
         if result == Won then { money1 random(60) + 60; text 7; gain money1; fame 1; exit }
         if result == Fled then { rejoin; text 8; hours 3; exit }
         text 9; hours 3; exit
