# The lantern-bearer: a link-boy lights you home, and stops dead at the hangman's door.
name     The lantern-bearer
card     $NnLnk00
picture  XNMAIN
where    street
when     night
weight   2

[start]
do       money1 2 + size; money2 60
block    0
chosen   bestattrskill(Per, Streetwise)
option 0                                   # hire him
  need   money >= money1
  chance none
  ok     pay money1; goto hangman
option 1                                   # buy his lantern outright
  need   money >= money2
  chance none
  ok     text 2; pay money2; additem leader, 0x3d; exit
option 2                                   # find your own way in the dark
  who    bestattrskill(Per, Streetwise)
  chance clamp(5, 90, attr(who, Per) + skill(who, Streetwise) / 2 - 10)
  ok     chosen who; text 3; practice who, Streetwise, 1, 10, 0; hours 1; exit
  fail   chosen who; text 4; injure who, 1, 1, 5; practice who, Streetwise, 0, 10, 0; hours 2; exit
option 3                                   # send him off
  chance none
  ok     text 5; exit

[hangman]
block    1
option 0                                   # let the hangman tend your hurts
  need   wounds(End) + wounds(Str) > 0
  chance none
  ok     text 6; heal party, End, 60; heal party, Str, 40; reputation -1, -3; hours 1; exit
option 1                                   # sit a while and hear him out
  who    bestattrskill(Cha, Streetwise)
  chance clamp(10, 90, attr(who, Cha) + skill(who, Streetwise) / 2)
  ok     chosen who; text 7; practice who, Streetwise, 1, 20, 0; attr party, DF, 2; hours 1; exit
  fail   chosen who; text 8; practice who, Streetwise, 0, 10, 0; hours 1; exit
option 2                                   # follow the boy the long way round
  chance none
  ok     text 9; hours 1; exit
