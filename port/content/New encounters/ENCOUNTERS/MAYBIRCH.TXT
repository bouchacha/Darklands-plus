# The May birch: village youths come singing at dawn to cut a May tree, and the finest birch in the wood stands over your tent.
name     The May birch
card     $NkMay00
picture  XTIPSEY
where    camp
when     any
weight   1
only     month >= 5 and month <= 6 and money >= 48   # the May trees and the Whitsun birches

[start]
block    0
chosen   bestattr(Cha)
option 0                                   # fell it for them, since you are the ones with the axes (Woodwise)
  who    bestskill(Woodwise)
  chance clamp(10, 90, skill(who, Woodwise) + attr(who, Str) / 3 + 10)
  ok     chosen who; text 1; hours 3; attr party, End, 2; reputation 1, 2; practice who, Woodwise, 1, 10, 0; exit
  fail   chosen who; text 2; hours 3; attr party, End, -2; practice who, Woodwise, 0, 10, 0; exit
option 1                                   # sing and dance with them round the birch first (Charisma, Agility)
  who    bestattr(Cha)
  chance clamp(10, 90, attr(who, Cha) + attr(who, Agl) / 3)
  ok     chosen who; money1 min(money, 24 + size * 12); text 3; pay money1; hours 3; reputation 1, 3; attr party, End, 2; exit
  fail   chosen who; text 4; hours 3; exit
option 2                                   # send them off to find another tree
  chance none
  ok     text 5; reputation -1, -2; exit
option 3                                   # pack up and leave them to it
  chance none
  ok     text 6; exit
