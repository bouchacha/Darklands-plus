# The winter dead: a grandmother kept frozen in the woodshed till the thaw, and a family that fears she walks.
name     The winter dead
card     $NhWdd00
picture  XGRV1
where    wild
terrain  mountains hills
when     day
weight   1
only     month == 12 or month <= 2

[start]
block    0
chosen   bestskill(Healing)
saints   0x39, 0x60, 0x55, -1            # Gertrude of Nivelles, Michael, Lazarus
option 0                                   # look at the body (Healing)
  who    bestskill(Healing)
  chance clamp(10, 90, skill(who, Healing) + attr(who, Per) / 3 - 5)
  ok     chosen who; text 1; hours 1; attr party, DF, 2; practice who, Healing, 1, 10, 0; exit
  fail   chosen who; text 2; hours 1; practice who, Healing, 0, 10, 0; exit
option 1                                   # keep watch in the shed tonight (one time in four she does rise)
  chance assured
  ok     if roll(25) goto risen
         text 3; until 6; attr party, End, -2; attr party, DF, 2; practice party, Virtue, 1, 10, 0; exit
option 2                                   # carry her down to the churchyard (endurance)
  who    party
  chance clamp(10, 90, avgattr(End) + avgattr(Str) / 2 - 5)
  ok     text 4; hours 8; attr party, End, -4; attr party, DF, 3; reputation 1, 3; exit
  fail   chosen slowest; text 5; hours 10; attr party, End, -4; injure slowest, 1, 1, 10; attr party, DF, 3; exit
option 3 saint
  ok     text 6; heal party, End, 30; exit
  fail   text 7
option 4                                   # go on
  chance none
  ok     text 8; exit

[risen]
picture  XGHOST
do       text 9; battle Clearing, Undead, lore / 4 + 1, 1
         if result == Won then { text 10; until 6; fame 1; attr party, DF, 2; exit }
         if result == Fled then { rejoin; text 11; until 6; exit }
         text 12; until 7; attr party, End, -3; exit
