# The shut gate: a physician of the Jews' lane, a gate locked for the night, and a choking child.
name     The shut gate
card     $NnJud00
picture  XPORTCUL
where    street
when     night
weight   1

[start]
do       money1 12
block    0
chosen   bestattr(Cha)
option 0                                   # reason with the gatekeeper
  who    bestattr(Cha)
  chance clamp(5, 90, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5 + standing / 4 - 10)
  ok     chosen who; text 1; reputation 1, 3; attr party, DF, 2; practice who, SpeakCommon, 1, 10, 0; hours 2; exit
  fail   chosen who; text 2; practice who, SpeakCommon, 0, 10, 0; disable 0
option 1                                   # pay him to see sense
  need   money >= money1
  who    bestskill(Streetwise)
  chance clamp(10, 90, 40 + skill(who, Streetwise) / 2)
  ok     chosen who; text 3; pay money1; reputation 1, 2; attr party, DF, 1; practice who, Streetwise, 1, 10, 0; hours 2; exit
  fail   chosen who; text 4; pay money1; practice who, Streetwise, 0, 10, 0; disable 1
option 2                                   # carry his remedy to the child yourselves (Healing, Latin)
  who    bestskill(Healing)
  chance clamp(5, 90, skill(who, Healing) + skill(who, Latin) / 3 + attr(who, Int) / 4 - 5)
  ok     chosen who; text 5; reputation 1, 3; attr party, DF, 2; practice who, Healing, 1, 20, 0; practice who, Latin, 1, 10, 0; hours 2; exit
  fail   chosen who; text 6; practice who, Healing, 0, 10, 0; hours 3; exit
option 3                                   # go on your way
  chance none
  ok     text 7; exit
