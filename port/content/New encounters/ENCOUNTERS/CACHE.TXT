# The robbers' cache: a robbers' sign cut in a stone, and a hollow beneath it.
name     The robbers' cache
card     $NwCac00
picture  XTRSCHST
where    wild
terrain  woods forest pinewood hills
when     day
weight   1
vars     field, dug

[start]
do       set field = Wilds; set dug = 0
block    0
chosen   spotter
option 0                                   # search for the hollow and dig it up
  who    spotter
  chance clamp(10, 90, maxskill(Woodwise) + maxattr(Per) / 2)
  ok     chosen who; set dug = 1; money1 random(96) + 48; text 1; gain money1; practice party, Woodwise, 1, 10, 0; hours 1
         if roll(40) then { text 3; set field = Surrounded; goto fight }
         exit
  fail   chosen who; text 2; practice party, Woodwise, 0, 10, 0; hours 2; disable 0
option 1                                   # lie in wait for the robbers
  who    party
  chance clamp(5, 90, avgskill(Stealth) + avgskill(Woodwise) / 2 + avgagility / 3)
  ok     text 4; if hour < 17 then until 18 else hours 1
         practice party, Stealth, 1, 10, 0; goto fight
  fail   text 5; if hour < 17 then until 18 else hours 1
         practice party, Stealth, 0, 10, 0; set field = Surrounded; goto fight
option 2                                   # mark it and tell the bailiff in town
  chance none
  ok     money1 36; text 6; gain money1; reputation 1, 3; hours 3; tocity
option 3                                   # leave it
  chance none
  ok     text 7; exit

[fight]
do       battle field, Bandit, lore / 4 + 1, clamp(size, 7, random(4) + lore / 3 + 1), BanditSergeant, lore % 3 + 1, 1
         if result == Won and dug then { money1 random(60) + 24; text 8; gain money1; hours 1; reputation 1, 2; fame 1; exit }
         if result == Won then { money1 random(96) + 72; text 11; gain money1; hours 1; reputation 1, 2; fame 1; exit }
         if result == Fled then { rejoin; text 9; hours 1; exit }
         loseitems party; text 10; hours 1; exit
