# The adder in the boot: a viper crept into a boot out of the night chill, and is in no hurry to leave.
name     The adder in the boot
card     $NkAdd00
picture  XLANDHO
where    camp
when     any
weight   1
only     month >= 4 and month <= 9             # adders sleep through the winter
vars     bit                                   # who was bitten

[start]
block    0
chosen   worstattr(Per)                        # the boot's owner
option 0                                   # tip the boot over slowly with a stick (Woodwise, Agility)
  who    bestskill(Woodwise, chosen)
  chance clamp(10, 90, skill(who, Woodwise) + attr(who, Agl) / 2 + 10)
  ok     chosen2 who; text 1; practice who, Woodwise, 1, 10, 0; exit
  fail   chosen2 who; set bit = who; text 2; practice who, Woodwise, 0, 10, 0; goto bitten
option 1                                   # strike off its head as it comes out (Edged)
  who    bestskill(Edged, chosen)
  chance clamp(5, 90, skill(who, Edged) + attr(who, Agl) / 3 - 10)
  ok     chosen2 who; text 3; attr party, End, 1; practice who, Edged, 1, 10, 0; exit
  fail   chosen2 who; set bit = chosen; text 4; practice who, Edged, 0, 10, 0; goto bitten
option 2                                   # sit still and let the sun draw it out (time)
  chance assured
  ok     text 5; hours 2; exit

[bitten]
block    6
chosen   bit
saints   0x69, 0x68, 0x6f, -1                  # Paul (the viper of Malta), Patrick, Raphael
items    0x73, 0x70                            # Essence o'Grace, New-wind
option 0                                   # cut, suck and bind the bite (Healing)
  who    bestskill(Healing, chosen)
  chance clamp(10, 90, skill(who, Healing) + attr(who, Int) / 3)
  ok     chosen2 who; text 7; attr chosen, End, -3; hours 1; practice who, Healing, 1, 10, 0; exit
  fail   chosen2 who; text 8; injure chosen, 1, 1, 10; hours 3; practice who, Healing, 0, 10, 0; exit
option 1 potion                            # a healing draught
  potion 0x73, clamp(0, 99, 60 + q / 2)
  potion 0x70, clamp(0, 99, 30 + q / 2)
  ok     text 9; hours 1; exit
  fail   text 10
option 2 saint
  ok     text 11; exit
  fail   text 12
option 3                                   # wrap them in blankets and let the poison run its course
  chance assured
  ok     text 13; injure chosen, 1, 1, 10; attr chosen, End, -3; hours 5; exit
