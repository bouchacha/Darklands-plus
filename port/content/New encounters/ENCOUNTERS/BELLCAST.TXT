# The bell-founder's boy: the bronze is ready, the master is in the tavern, and his apprentice dare not pour.
name     The bell-founder's boy
card     $NdBel00
picture  XBSMOKE
where    street
when     day
weight   1
only     month >= 4 and month <= 10

[start]
block    0
chosen   bestskill(Artifice)
saints   0x01, 0x0d                      # Agatha (patron of bell-founders), Barbara
option 0                                   # help him tap the furnace and pour
  who    bestskill(Artifice)
  chance clamp(5, 90, skill(who, Artifice) + attr(who, Str) / 3 - 10)
  ok     chosen who; text 1; practice who, Artifice, 1, 10, 0; goto master
  fail   chosen who; text 2; injure who, 1, 1, 10; practice who, Artifice, 0, 10, 0; goto master
option 1                                   # run and fetch the master from the tavern
  who    bestskill(Streetwise)
  chance clamp(5, 90, skill(who, Streetwise) + attr(who, Per) / 3 + 5)
  ok     chosen who; text 3; reputation 1, 2; practice who, Streetwise, 1, 10, 0; hours 1; exit
  fail   chosen who; text 4; practice who, Streetwise, 0, 10, 0; hours 2; exit
option 2                                   # judge the metal for yourselves
  who    bestskill(Alchemy)
  chance clamp(5, 90, skill(who, Alchemy) + attr(who, Int) / 3 - 5)
  ok     chosen who; text 5; reputation 1, 2; practice who, Alchemy, 1, 10, 0; hours 1; exit
  fail   chosen who; text 6; practice who, Alchemy, 0, 10, 0; goto master
option 3 saint
  ok     text 7; reputation 1, 3; exit
  fail   text 8
option 4                                   # walk on
  chance none
  ok     text 9; exit

[master]
block    10
option 0                                   # take the hammer from him
  who    bestskill(Impact)
  chance clamp(5, 90, skill(who, Impact) + attr(who, Str) / 3 - 5)
  ok     chosen who; text 11; reputation 1, 2; practice who, Impact, 1, 10, 0; exit
  fail   chosen who; text 12; injure who, 1, 1, 10; practice who, Impact, 0, 10, 0; exit
option 1                                   # reason with him
  who    bestattrskill(Cha, Religion)
  chance clamp(5, 90, attr(who, Cha) + skill(who, Religion) / 2 - 10)
  ok     chosen who; text 13; attr party, DF, 2; reputation 1, 2; exit
  fail   chosen who; text 14; exit
option 2                                   # leave them to it
  chance none
  ok     text 15; exit
