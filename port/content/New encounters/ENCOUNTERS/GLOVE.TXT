# The lady's glove: a noblewoman lets fall a glove from her litter.
name     The lady's glove
card     $NsGlv00
picture  XPRTYLDY
where    street
when     day
weight   1
vars     thief

[start]
block    0
chosen   bestattr(Cha)
option 0                                   # return it with your best courtesy
  who    bestattr(Cha)
  chance clamp(5, 90, (2 * attr(who, Cha) + skill(who, SpeakCommon)) / 3 + min(20, fame / 10) - 5)
  ok     chosen who; text 1; fame 1; reputation 1, 2; practice who, SpeakCommon, 1, 10, 0; exit
  fail   chosen who; text 2; practice who, SpeakCommon, 0, 10, 0; goto guards
option 1                                   # slip it into your sleeve
  who    bestskill(Stealth)
  chance clamp(5, 90, skill(who, Stealth) + attr(who, Agl) / 3)
  ok     chosen who; money1 random(36) + 36; text 3; gain money1; attr party, DF, -2; practice who, Stealth, 1, 10, 0; exit
  fail   chosen who; set thief = 1; text 4; practice who, Stealth, 0, 10, 0; goto guards
option 2                                   # let it lie
  chance none
  ok     text 5; exit

[guards]
picture  GUARD1
block    6
option 0                                   # kneel and hold out the glove
  chance assured
  ok     text 7; injure chosen, 1, 1, 5; reputation -1, -2; exit
option 1                                   # stand your ground and fight
  chance combat
  ok     goto fight

[fight]
do       battle MainStreet, Guard, lore / 4 + 1, 4
         if result == Won and thief then { text 8; reputation -6, -12; hours 1; exit }
         if result == Won then { text 8; reputation -1, -3; hours 1; exit }
         if result == Fled and thief then { rejoin; text 9; reputation -3, -6; exit }
         if result == Fled then { rejoin; text 9; reputation -1, -3; exit }
         text 10; hours 4
         if thief then reputation -2, -4 else reputation -1, -2
         exit
