# The hunting lodge: a lord's lost goshawk, and the welcome at his hunting lodge for whoever brings it back.
name     The hunting lodge
card     $NfLdg00
picture  XBUDLGHT
where    wild
terrain  forest woods
when     day
weight   1

[start]
block    0
chosen   bestattr(Cha)
option 0                                   # coax her down (Charisma, Agility, Woodwise)
  who    bestattrsum(Cha, Agl)
  chance clamp(5, 90, (attr(who, Cha) + attr(who, Agl)) / 2 + skill(who, Woodwise) / 4 - 5)
  ok     chosen who; practice who, Woodwise, 1, 5, -1; hours 1; goto lodge
  fail   chosen who; text 1; hours 1; exit
option 1                                   # net her in a cloak and sell her (Agility, Streetwise)
  who    bestattr(Agl)
  chance clamp(5, 85, attr(who, Agl) + skill(who, Streetwise) / 3 - 15)
  ok     chosen who; money1 96 + random(96); text 2; gain money1; attr party, DF, -2; practice who, Streetwise, 1, 10, 0; exit
  fail   chosen who; text 3; injure who, 1, 1, 5; exit
option 2                                   # leave her
  chance none
  ok     text 4; exit

[lodge]
picture  XVC1
do       money1 24 + size * 12
block    5
option 0                                   # ride with the hunt (Riding)
  need   mounted
  who    party
  chance clamp(10, 90, avgskill(Riding) + avgagility / 2 - 5)
  ok     text 6; fame 1; attr party, End, 2; practice party, Riding, 1, 10, 0; hours 6; exit
  fail   chosen worstskill(Riding); text 7; injure chosen, 1, 1, 5; practice party, Riding, 0, 10, 0; hours 6; exit
option 1                                   # take a boar-spear with the huntsmen (Polearms, Strength)
  who    bestskill(Polearms)
  chance clamp(5, 90, skill(who, Polearms) + attr(who, Str) / 3 - 10)
  ok     chosen who; text 8; reputation 1, 3; practice who, Polearms, 1, 10, 0; hours 6; exit
  fail   chosen who; text 9; injure who, 1, 1, 10; practice who, Polearms, 0, 10, 0; hours 6; exit
option 2                                   # sit at his table and tell him the news (Charisma, Speak Common)
  who    bestattr(Cha)
  chance clamp(5, 90, (4 * attr(who, Cha) + skill(who, SpeakCommon)) / 5 + min(15, fame / 10))
  ok     chosen who; text 10; reputation 1, 3; practice who, SpeakCommon, 1, 10, 0; until 7; attr party, End, 3; exit
  fail   chosen who; text 11; practice who, SpeakCommon, 0, 10, 0; until 7; attr party, End, 3; exit
option 3                                   # take his reward and go
  chance none
  ok     text 12; gain money1; exit
