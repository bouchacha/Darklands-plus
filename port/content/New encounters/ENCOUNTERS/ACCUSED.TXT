# The witch-finder's prisoner: zealots drag an old woman to the stake without a trial.
name     The witch-finder's prisoner
card     $NwAcc00
picture  XMOB
where    wild
terrain  farmland plains road woods
when     day
weight   1

[start]
do       money1 clamp(48, 480, money / 6)
block    0
chosen   bestskill(Religion)
option 0                                   # plead for a trial before the bishop's court
  who    bestskill(Religion)
  chance clamp(5, 90, skill(who, Religion) + attr(who, Cha) / 3 + skill(who, Latin) / 4 - 15)
  ok     chosen who; text 1; practice who, Religion, 1, 10, 0; attr party, DF, 3; hours 1; exit
  fail   chosen who; text 2; practice who, Religion, 0, 10, 0; disable 0
option 1                                   # buy her freedom
  need   money >= money1
  chance none
  ok     text 3; pay money1; attr party, DF, 2; additem leader, 0x56; exit
option 2                                   # fall upon the zealots
  chance combat
  ok     text 4; goto fight
option 3                                   # turn away
  chance none
  ok     text 5; exit

[fight]
do       battle Clearing, Follower, lore / 4 + 1, clamp(size, 6, random(4) + lore / 3 + 1), Zealot, lore % 3 + 1, 1
         if result == Won then { text 6; hours 1; attr party, DF, 3; practice party, Virtue, 1, 10, 0; fame 1; exit }
         if result == Fled then { rejoin; text 7; hours 1; exit }
         text 8; hours 3; exit
