# The broken mill-dam: a miller's dam bursting after heavy rain.
name     The broken mill-dam
card     $NwMil00
picture  XSTMTIDE
where    wild
terrain  farmland plains woods marsh road
when     day
weight   1
only     month >= 3 and month <= 11

[start]
block    0
option 0                                   # put your shoulders to it (strength)
  who    bestattr(Str)
  chance clamp(10, 90, attr(who, Str) + avgattr(Str) / 2 + avgattr(End) / 4 - 15)
  ok     text 1; hours 3; attr party, End, -3; reputation 1, 3; exit
  fail   chosen who; text 2; hours 3; injure who, 1, 1, 10; keepitems who, clamp(1, 95, attr(who, Agl) + 50); exit
option 1                                   # brace it with timbers (craft)
  who    bestskill(Artifice)
  chance clamp(10, 90, skill(who, Artifice) + attr(who, Int) / 3)
  ok     chosen who; text 3; hours 2; practice who, Artifice, 1, 10, 0; reputation 1, 3; exit
  fail   chosen who; text 4; hours 2; attr party, End, -2; practice who, Artifice, 0, 10, 0; exit
option 2                                   # send for help (quicker on horseback)
  who    bestskill(Riding)
  chance clamp(10, 90, 30 + mounted * 25 + skill(who, Riding) / 2 + attr(who, Agl) / 4)
  ok     chosen who; text 5; hours 4; reputation 1, 2
         if mounted then practice who, Riding, 1, 10, 0
         exit
  fail   chosen who; text 6; hours 5
         if mounted then practice who, Riding, 0, 10, 0
         exit
option 3                                   # leave them to it
  chance none
  ok     text 7; exit
